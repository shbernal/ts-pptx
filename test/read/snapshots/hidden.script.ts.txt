/*
 * Generated from a .pptx by ts-pptx/script.
 *
 * Template-anchored: the source deck is reused as the template, so its slide masters,
 * layouts, theme and document properties are the originals, byte for byte. Only the
 * slide content below was rebuilt through the public write API.
 *
 * Expects ./template.pptx (the source deck, unmodified) beside this file.
 * Writes ./output.pptx. Needs an ESM context — it uses top-level await.
 *
 * FIDELITY: 3 declared loss(es). Everything not listed here is expected to
 * survive unchanged; a difference that is not in this list is a defect, not a caveat.
 *
 * Slide 1:
 * - text.color.inherited (replaceText) [flattened, unsupported] — this run inherits its colour from the
 *   placeholder, master or theme; the write path paints an uncoloured run black instead of leaving it to
 *   inherit, so the inherited colour is resolved and baked in and no longer tracks a theme change
 * - text.color.inherited (setText) [flattened, unsupported] — this run inherits its colour from the placeholder,
 *   master or theme; the write path paints an uncoloured run black instead of leaving it to inherit, so the
 *   inherited colour is resolved and baked in and no longer tracks a theme change
 * Slide 2:
 * - text.color.inherited (replaceTextBullet1) [flattened, unsupported] — this run inherits its colour from the
 *   placeholder, master or theme; the write path paints an uncoloured run black instead of leaving it to
 *   inherit, so the inherited colour is resolved and baked in and no longer tracks a theme change
 */

import { readFile, writeFile } from 'node:fs/promises'
import { fileURLToPath } from 'node:url'
import TsPptx from '@shbernal/ts-pptx'
import { Presentation } from '@shbernal/ts-pptx/read'

/** Resolve a path against this script rather than against the working directory. */
const here = (name: string): string => fileURLToPath(new URL(name, import.meta.url))

// The source deck is the template: fromTemplate strips its slides and leaves its masters,
// layouts, theme and document properties exactly as they were.
const deck = await Presentation.fromTemplate(here('./template.pptx'))

/**
 * A generator sized to the template. appendSlides compares slide sizes exactly and
 * throws when they differ, so the inches here must round-trip to the source EMU.
 */
function generator(): TsPptx {
	const pptx = new TsPptx()
	pptx.defineLayout({ name: 'source', width: 13.333333, height: 7.5 })
	pptx.layout = 'source'
	return pptx
}
